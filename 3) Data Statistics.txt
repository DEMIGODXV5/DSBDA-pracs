# -----------------------------------------------------
# 6. Basic Statistical Details of Each Species
# -----------------------------------------------------

species_list = [
    'setosa',
    'versicolor',
    'virginica'
]

for species in species_list:
    print("\n========================") 
    print("Species:", species) 
    print("========================")
    data = df[df['species'] == species] 
"""here the data contains onyl the rows for the particular species.
    df['species'] == species creates a True/False condition by checking matching species values.
    Outer df[...] uses this condition to filter and select only matching rows.
    Result is a new dataframe containing data of that particular species only.
"""

    print("\nMean:")
    print(data.mean(numeric_only=True)) 
    """Pandas automatically:
              selects numeric columns
              calculates mean for EACH numeric column separately"""

    print("\nMedian:")
    print(data.median(numeric_only=True))

    print("\nMinimum:")
    print(data.min(numeric_only=True))

    print("\nMaximum:")
    print(data.max(numeric_only=True))

    print("\nStandard Deviation:")
    print(data.std(numeric_only=True))

    print("\nVariance:")
    print(data.var(numeric_only=True))
