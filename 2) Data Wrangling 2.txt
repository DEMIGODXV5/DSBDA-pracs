
# -----------------------------------------------------
# 1. Import Required Libraries
# -----------------------------------------------------
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt


# -----------------------------------------------------
# 2. Load Dataset
# -----------------------------------------------------

df = pd.read_csv("https://github.com/wise-venture18/dsbdal/raw/refs/heads/main/2.DataWrangling2/StudentsPerformance.csv")
print("First 5 Rows:")
print(df.head())


# -----------------------------------------------------
# 3. Dataset Information
# -----------------------------------------------------

print(df.info())


print(df.shape)
df.describe()


# -----------------------------------------------------
# 4. Check Missing Values
# -----------------------------------------------------

print("\nMissing Values:")
print(df.isnull().sum())

# isnull() checks missing values


# -----------------------------------------------------
# 5. Handle Missing Values
# -----------------------------------------------------

# Fill numeric column with mean

df['math score'] = df['math score'].fillna(
    df['math score'].mean()
)

# Fill categorical column with mode

df['gender'] = df['gender'].fillna(
    df['gender'].mode()[0]
)

print("\nMissing Values After Handling:")
df.isnull().sum()


# -----------------------------------------------------
# 6. Check Inconsistencies
# -----------------------------------------------------

print("\nUnique Gender Values:")
print(df['gender'].unique())

# Convert values to lowercase

df['gender'] = df['gender'].str.lower()

print("\nGender Values After Cleaning:")
print(df['gender'].unique())

# unique() checks inconsistent values


# -----------------------------------------------------
# 7. Statistical Summary after handling missing values and inconsistencies
# -----------------------------------------------------

print("\nStatistical Summary:")
print(df.describe())


# -----------------------------------------------------
# 8. Boxplot Before Outlier Removal
# -----------------------------------------------------

plt.boxplot(df['math score'])
plt.title("Boxplot of Math Score Before Removing Outliers")
plt.ylabel("Math Score")
plt.show()

# Boxplot helps detect outliers


# -----------------------------------------------------
# 9. Detect Outliers using IQR Method
# -----------------------------------------------------

Q1 = df['math score'].quantile(0.25)

Q3 = df['math score'].quantile(0.75)

IQR = Q3 - Q1

lower_limit = Q1 - 1.5 * IQR

upper_limit = Q3 + 1.5 * IQR

print("\nLower Limit:", lower_limit)

print("Upper Limit:", upper_limit)


# Detect outliers

outliers = df[
    (df['math score'] < lower_limit) |
    (df['math score'] > upper_limit)
]

print("\nOutliers:")

print(outliers[['math score']])


# -----------------------------------------------------
# 10. Remove Outliers
# -----------------------------------------------------

df = df[
    (df['math score'] >= lower_limit) &
    (df['math score'] <= upper_limit)
]

print("\nShape After Removing Outliers:")

print(df.shape)


# -----------------------------------------------------
# 11. Boxplot After Outlier Removal
# -----------------------------------------------------

plt.boxplot(df['math score'])
plt.title("Boxplot After Removing Outliers")
plt.ylabel("Math Score")
plt.show()


# -----------------------------------------------------
# 12. Data Transformation
# -----------------------------------------------------

# Apply Log Transformation

df['math_log'] = np.log1p(df['math score'])
print("\nTransformed Data:")
print(df[['math score', 'math_log']].head())

# log1p() reduces skewness


# -----------------------------------------------------
# 13. Histogram After Transformation
# -----------------------------------------------------

plt.hist(df['math_log'])
plt.title("Log Transformed Math Score")
plt.xlabel("math_log")
plt.ylabel("Frequency")
plt.show()


# -----------------------------------------------------
# 14. Final Dataset
# -----------------------------------------------------

print("\nFinal Dataset:")

print(df.head())
df.describe()
