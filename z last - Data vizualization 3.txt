import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns



df = pd.read_csv("https://github.com/wise-venture18/dsbdal/raw/refs/heads/main/10.DataVisualization3/iris.csv")
df.head()
df.info()
df.describe()


#Features and Their Types
print("\nNumber of Features:")
print(df.shape[1])
print("\nData Types:")
print(df.dtypes)


#Separate Numeric and Nominal Features
numeric_features = df.select_dtypes(include=['float64', 'int64']).columns
nominal_features = df.select_dtypes(include=['object']).columns

print("Numeric Features:")
print(numeric_features)

print("\nNominal Features:")
print(nominal_features)



#Histogram for Each Feature
df[numeric_features].hist(figsize=(10, 8), edgecolor='black')
plt.suptitle("Histogram of Iris Features")
plt.show()


#Combined Boxplot
plt.figure(figsize=(10, 6))
sns.boxplot(data=df[numeric_features])
plt.title("Boxplot of Iris Features")
plt.show()


#Individual Boxplots
for col in numeric_features:
    plt.figure(figsize=(5, 4))
    
    sns.boxplot(y=df[col])
    
    plt.title(f"Boxplot of {col}")
    plt.show()


#Outlier Detection
print("Outlier Detection:\n")
for col in numeric_features:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1

    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR

    outliers = df[(df[col] < lower_bound) | (df[col] > upper_bound)]

    print(f"{col}: {len(outliers)} outliers detected")



print("Inference:")

print("1. Sepal length and sepal width show moderate spread in values.")
print("2. Petal length and petal width show clearer separation and wider variation.")
print("3. Histograms indicate petal features have clustered distributions.")
print("4. Boxplots help identify extreme values (outliers) in some features.")
print("5. Petal measurements are more useful for distinguishing flower species.")
